    class SoundEffects {
      constructor() {
        this.ctx = null;
      }

      init() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioCtx();
        }
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      playBlowSound(duration = 1.6) {
        this.init();
        if (!this.ctx) return;

        const bufferSize = this.ctx.sampleRate * duration;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);

        let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
        for (let i = 0; i < bufferSize; i++) {
          const white = Math.random() * 2 - 1;
          b0 = 0.99886 * b0 + white * 0.0555179;
          b1 = 0.99332 * b1 + white * 0.0750759;
          b2 = 0.96900 * b2 + white * 0.1538520;
          b3 = 0.86650 * b3 + white * 0.3104856;
          b4 = 0.55000 * b4 + white * 0.5329522;
          b5 = -0.7616 * b5 - white * 0.0168980;
          data[i] = (b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362) * 0.12;
          b6 = white * 0.115926;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.setValueAtTime(450, this.ctx.currentTime);
        filter.frequency.exponentialRampToValueAtTime(120, this.ctx.currentTime + duration);

        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(0.01, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0.5, this.ctx.currentTime + 0.2);
        gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.ctx.destination);

        noise.start();
      }

      playCelebrateSound() {
        this.init();
        if (!this.ctx) return;

        const notes = [523.25, 659.25, 783.99, 1046.50, 1318.51, 1567.98];
        notes.forEach((freq, index) => {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();

          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq, this.ctx.currentTime + index * 0.09);

          gain.gain.setValueAtTime(0, this.ctx.currentTime + index * 0.09);
          gain.gain.linearRampToValueAtTime(0.18, this.ctx.currentTime + index * 0.09 + 0.04);
          gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + index * 0.09 + 1.2);

          osc.connect(gain);
          gain.connect(this.ctx.destination);

          osc.start(this.ctx.currentTime + index * 0.09);
          osc.stop(this.ctx.currentTime + index * 0.09 + 1.3);
        });
      }
    }

      function animate() {

            flame.rotation.x = Math.cos(elapsedTime * 2.5 + index) * 0.08;

            if (light) light.intensity = 2.5 + flicker * 1.5;
          } else if (appState === 'BLOWING') {
            flame.scale.x = Math.max(0, flame.scale.x - delta * 2.5);
            flame.scale.y = Math.max(0, flame.scale.y - delta * 2.5);
            flame.scale.z = Math.max(0, flame.scale.z - delta * 2.5);
            flame.rotation.z += delta * 15;

            if (light) light.intensity = Math.max(0, light.intensity - delta * 8);
          } else if (appState === 'EXTINGUISHED') {
            flame.scale.set(0, 0, 0);
            flame.visible = false;
            if (light) light.intensity = 0;
          }
        });

        // Smoke Particle Animation
        smokeSystems.forEach((system) => {
          if (appState === 'EXTINGUISHED') {
            system.material.opacity = 0.45;
            const posAttr = system.points.geometry.attributes.position;

            system.data.forEach((p, i) => {
              p.y += delta * p.speed;
              p.x += Math.sin(elapsedTime * 3 + i) * delta * 0.08;

              if (p.y > 0.8) {
                p.y = 0.35;
                p.x = (Math.random() - 0.5) * 0.05;
              }

              posAttr.setXYZ(i, p.x, p.y, p.z);
            });
            posAttr.needsUpdate = true;
          } else {
            system.material.opacity = 0;
          }
        });

        // Confetti physics animation
        confettiItems.forEach((c) => {
          if (appState === 'EXTINGUISHED') {
            c.mesh.visible = true;
            c.pos.x += c.vel.x * delta;
            c.pos.y += c.vel.y * delta;
            c.pos.z += c.vel.z * delta;

            c.mesh.rotation.x += c.rotSpeed.x * delta;
            c.mesh.rotation.y += c.rotSpeed.y * delta;

            if (c.pos.y < -1) {
              c.pos.set((Math.random() - 0.5) * 3.5, 2.5 + Math.random() * 1.5, (Math.random() - 0.5) * 3.5);
            }
            c.mesh.position.copy(c.pos);
          } else {
            c.mesh.visible = false;
          }
        });

        // Camera Lerp
        let targetPos = [0, 1.8, 4.2];
        if (cameraPreset === 'top') targetPos = [0, 3.8, 0.1];
        if (cameraPreset === 'close') targetPos = [0, 0.9, 1.8];

        camera.position.x += (targetPos[0] - camera.position.x) * delta * 2.5;
        camera.position.y += (targetPos[1] - camera.position.y) * delta * 2.5;
        camera.position.z += (targetPos[2] - camera.position.z) * delta * 2.5;

        controls.update();
        renderer.render(scene, camera);
      }

      animate();

      window.addEventListener('resize', () => {
        const w = window.innerWidth;
        const h = window.innerHeight;
        camera.aspect = w / h;
        camera.updateProjectionMatrix();
        renderer.setSize(w, h);
      });
    }

    function handlePrimaryAction() {
      if (appState === 'WAITING') {
        appState = 'BLOWING';
        sfx.playBlowSound(1.5);
        updateUI();

        setTimeout(() => {
          appState = 'EXTINGUISHED';
          sfx.playCelebrateSound();
          updateUI();
        }, 1100);
      } else if (appState === 'EXTINGUISHED') {
        appState = 'WAITING';
        updateUI();
      }
    }

      const headerWaiting = document.getElementById('header-waiting');
      const headerExtinguished = document.getElementById('header-extinguished');
      const btn = document.getElementById('btn-blow-reset');
      const btnIcon = document.getElementById('btn-icon');
      const btnLabel = document.getElementById('btn-label');

      if (appState === 'WAITING') {
        headerWaiting.classList.remove('hidden');
        headerExtinguished.classList.add('hidden');

        btn.className = "flex-1 py-3.5 px-6 rounded-2xl bg-gradient-to-r from-pink-500 via-rose-500 to-amber-500 hover:from-pink-600 hover:to-amber-600 text-white font-bold text-base md:text-lg shadow-lg shadow-pink-500/30 active:scale-95 transition-all flex items-center justify-center gap-2 border border-white/20";
        btnIcon.className = "fa-solid fa-wind animate-pulse";
        btnLabel.innerText = "Blow Candles";
        btn.disabled = false;
      } else if (appState === 'BLOWING') {
        btnIcon.className = "fa-solid fa-spinner animate-spin";
        btnLabel.innerText = "Blowing Candles...";
        btn.disabled = true;
      } else if (appState === 'EXTINGUISHED') {
        headerWaiting.classList.add('hidden');
        headerExtinguished.classList.remove('hidden');

        btn.className = "flex-1 py-3.5 px-6 rounded-2xl bg-gradient-to-r from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white font-bold text-base md:text-lg shadow-lg shadow-teal-500/30 active:scale-95 transition-all flex items-center justify-center gap-2 border border-white/20";
        btnIcon.className = "fa-solid fa-rotate-left";
        btnLabel.innerText = "Re-light Candles";
        btn.disabled = false;
      }
    }

    function setCameraPreset(preset) {
      cameraPreset = preset;

      const btns = {
        default: document.getElementById('cam-btn-default'),
        close: document.getElementById('cam-btn-close'),
        top: document.getElementById('cam-btn-top')
      };

      Object.keys(btns).forEach(key => {
        if (key === preset) {
          btns[key].className = "px-3 py-1.5 rounded-xl border transition bg-pink-500/20 border-pink-500 text-pink-300 font-semibold";
        } else {
          btns[key].className = "px-3 py-1.5 rounded-xl border transition bg-white/5 border-white/10 text-gray-400 hover:bg-white/10";
        }
      });
    }

    async function toggleMicrophone() {
      const micBtn = document.getElementById('btn-mic');
      const micIcon = document.getElementById('mic-icon');
      const meterContainer = document.getElementById('mic-meter-container');

      if (isMicActive) {
        stopMic();
      } else {
        try {
          micStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          audioCtx = new AudioContext();
          analyser = audioCtx.createAnalyser();
          analyser.fftSize = 256;

          const source = audioCtx.createMediaStreamSource(micStream);
          source.connect(analyser);

          isMicActive = true;
          micBtn.className = "p-3.5 rounded-2xl border transition-all flex items-center justify-center bg-red-500/20 border-red-500 text-red-400 animate-pulse";
          micIcon.className = "fa-solid fa-microphone text-xl";
          meterContainer.classList.remove('hidden');

          detectMicBlow();
        } catch (err) {
          console.error("Microphone access error:", err);
        }
      }
    }

    function stopMic() {
      if (animFrameId) cancelAnimationFrame(animFrameId);
      if (micStream) {
        micStream.getTracks().forEach(track => track.stop());
      }
      if (audioCtx) {
        audioCtx.close();
      }
      isMicActive = false;

      const micBtn = document.getElementById('btn-mic');
      const micIcon = document.getElementById('mic-icon');
      const meterContainer = document.getElementById('mic-meter-container');

      micBtn.className = "p-3.5 rounded-2xl border transition-all flex items-center justify-center bg-white/5 border-white/10 text-gray-300 hover:bg-white/10";
      micIcon.className = "fa-solid fa-microphone-slash text-xl";
      meterContainer.classList.add('hidden');
    }

    function detectMicBlow() {
      if (!analyser) return;

      const dataArray = new Uint8Array(analyser.frequencyBinCount);
      analyser.getByteFrequencyData(dataArray);

      let sum = 0;
      for (let i = 0; i < 15; i++) {
        sum += dataArray[i];
      }
      const average = sum / 15;

      const meterBar = document.getElementById('mic-meter-bar');
      if (meterBar) {
        meterBar.style.width = `${Math.min(100, (average / 80) * 100)}%`;
      }

      if (average > 65 && appState === 'WAITING') {
        handlePrimaryAction();
        stopMic();
        return;
      }

      animFrameId = requestAnimationFrame(detectMicBlow);
    }
